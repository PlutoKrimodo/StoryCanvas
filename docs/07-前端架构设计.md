## 3. 路由设计

### 3.1 路由配置

```typescript
// src/App.tsx
import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom';
import { useAuthStore } from './stores/authStore';

// 页面组件
import Login from './pages/auth/Login';
import Register from './pages/auth/Register';
import Home from './pages/home/Home';
import BookList from './pages/books/BookList';
import BookDetail from './pages/books/BookDetail';
import Editor from './pages/editor/Editor';
import Profile from './pages/profile/Profile';

// 路由守卫
const ProtectedRoute = ({ children }: { children: React.ReactNode }) => {
  const { isAuthenticated } = useAuthStore();
  return isAuthenticated ? <>{children}</> : <Navigate to="/login" />;
};

function App() {
  return (
    <BrowserRouter>
      <Routes>
        {/* 公开路由 */}
        <Route path="/login" element={<Login />} />
        <Route path="/register" element={<Register />} />
        
        {/* 受保护路由 */}
        <Route path="/" element={
          <ProtectedRoute>
            <Home />
          </ProtectedRoute>
        } />
        <Route path="/books" element={
          <ProtectedRoute>
            <BookList />
          </ProtectedRoute>
        } />
        <Route path="/books/:bookId" element={
          <ProtectedRoute>
            <BookDetail />
          </ProtectedRoute>
        } />
        <Route path="/editor/:bookId" element={
          <ProtectedRoute>
            <Editor />
          </ProtectedRoute>
        } />
        <Route path="/profile" element={
          <ProtectedRoute>
            <Profile />
          </ProtectedRoute>
        } />
        
        {/* 默认重定向 */}
        <Route path="*" element={<Navigate to="/" />} />
      </Routes>
    </BrowserRouter>
  );
}

export default App;
```

### 4.1 认证状态

```typescript
// src/stores/authStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import { User, AuthResponse } from '../types/auth';

interface AuthState {
  user: User | null;
  accessToken: string | null;
  refreshToken: string | null;
  isAuthenticated: boolean;
  
  login: (response: AuthResponse) => void;
  logout: () => void;
  updateUser: (user: Partial<User>) => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      user: null,
      accessToken: null,
      refreshToken: null,
      isAuthenticated: false,
      
      login: (response) => set({
        user: response.user,
        accessToken: response.access_token,
        refreshToken: response.refresh_token,
        isAuthenticated: true,
      }),
      
      logout: () => set({
        user: null,
        accessToken: null,
        refreshToken: null,
        isAuthenticated: false,
      }),
      
      updateUser: (userData) => set((state) => ({
        user: state.user ? { ...state.user, ...userData } : null,
      })),
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({
        user: state.user,
        accessToken: state.accessToken,
        refreshToken: state.refreshToken,
        isAuthenticated: state.isAuthenticated,
      }),
    }
  )
);
```

### 4.2 编辑器状态

```typescript
// src/stores/editorStore.ts
import { create } from 'zustand';
import { Canvas } from 'fabric';
import { Page, PageElement } from '../types/page';

interface EditorState {
  // 当前编辑的绘本
  bookId: string | null;
  
  // 页面相关
  pages: Page[];
  currentPageId: string | null;
  
  // 画布相关
  canvas: Canvas | null;
  selectedElement: PageElement | null;
  
  // 操作历史
  history: any[];
  historyIndex: number;
  
  // 操作方法
  setBookId: (bookId: string) => void;
  setPages: (pages: Page[]) => void;
  setCurrentPage: (pageId: string) => void;
  setCanvas: (canvas: Canvas) => void;
  setSelectedElement: (element: PageElement | null) => void;
  
  // 撤销/重做
  undo: () => void;
  redo: () => void;
  pushHistory: (state: any) => void;
  
  // 页面操作
  addPage: (page: Page) => void;
  updatePage: (pageId: string, data: Partial<Page>) => void;
  deletePage: (pageId: string) => void;
  reorderPages: (pageIds: string[]) => void;
  
  // 元素操作
  addElement: (element: PageElement) => void;
  updateElement: (elementId: string, data: Partial<PageElement>) => void;
  deleteElement: (elementId: string) => void;
}

export const useEditorStore = create<EditorState>()((set, get) => ({
  bookId: null,
  pages: [],
  currentPageId: null,
  canvas: null,
  selectedElement: null,
  history: [],
  historyIndex: -1,
  
  setBookId: (bookId) => set({ bookId }),
  
  setPages: (pages) => set({ pages }),
  
  setCurrentPage: (pageId) => set({ currentPageId: pageId }),
  
  setCanvas: (canvas) => set({ canvas }),
  
  setSelectedElement: (element) => set({ selectedElement: element }),
  
  undo: () => {
    const { historyIndex, history, canvas } = get();
    if (historyIndex > 0) {
      const newIndex = historyIndex - 1;
      canvas?.loadFromJSON(history[newIndex], () => {
        canvas.renderAll();
      });
      set({ historyIndex: newIndex });
    }
  },
  
  redo: () => {
    const { historyIndex, history, canvas } = get();
    if (historyIndex < history.length - 1) {
      const newIndex = historyIndex + 1;
      canvas?.loadFromJSON(history[newIndex], () => {
        canvas.renderAll();
      });
      set({ historyIndex: newIndex });
    }
  },
  
  pushHistory: (state) => {
    const { historyIndex, history } = get();
    const newHistory = history.slice(0, historyIndex + 1);
    newHistory.push(state);
    set({
      history: newHistory,
      historyIndex: newHistory.length - 1,
    });
  },
  
  addPage: (page) => set((state) => ({
    pages: [...state.pages, page],
  })),
  
  updatePage: (pageId, data) => set((state) => ({
    pages: state.pages.map((p) =>
      p.id === pageId ? { ...p, ...data } : p
    ),
  })),
  
  deletePage: (pageId) => set((state) => ({
    pages: state.pages.filter((p) => p.id !== pageId),
    currentPageId: state.currentPageId === pageId
      ? state.pages.find((p) => p.id !== pageId)?.id || null
      : state.currentPageId,
  })),
  
  reorderPages: (pageIds) => set((state) => ({
    pages: pageIds.map((id) => state.pages.find((p) => p.id === id)!),
  })),
  
  addElement: (element) => set((state) => {
    const currentPage = state.pages.find((p) => p.id === state.currentPageId);
    if (currentPage) {
      return {
        pages: state.pages.map((p) =>
          p.id === state.currentPageId
            ? { ...p, elements: [...p.elements, element] }
            : p
        ),
      };
    }
    return state;
  }),
  
  updateElement: (elementId, data) => set((state) => ({
    pages: state.pages.map((p) =>
      p.id === state.currentPageId
        ? {
            ...p,
            elements: p.elements.map((e) =>
              e.id === elementId ? { ...e, ...data } : e
            ),
          }
        : p
    ),
  })),
  
  deleteElement: (elementId) => set((state) => ({
    pages: state.pages.map((p) =>
      p.id === state.currentPageId
        ? {
            ...p,
            elements: p.elements.filter((e) => e.id !== elementId),
          }
        : p
    ),
    selectedElement: state.selectedElement?.id === elementId
      ? null
      : state.selectedElement,
  })),
}));
```

## 5. API 请求封装

### 5.1 Axios 实例

```typescript
// src/api/client.ts
import axios, { AxiosInstance, AxiosError, InternalAxiosRequestConfig } from 'axios';
import { useAuthStore } from '../stores/authStore';
import { message } from 'antd';

// 创建 Axios 实例
const client: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || 'http://localhost:8080/api/v1',
  timeout: 30000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// 请求拦截器
client.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const { accessToken } = useAuthStore.getState();
    if (accessToken) {
      config.headers.Authorization = `Bearer ${accessToken}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 响应拦截器
client.interceptors.response.use(
  (response) => response,
  async (error: AxiosError) => {
    const originalRequest = error.config as any;
    
    // 处理 401 错误（Token 过期）
    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;
      
      try {
        const { refreshToken, login, logout } = useAuthStore.getState();
        
        if (refreshToken) {
          // 尝试刷新 Token
          const response = await axios.post(
            `${import.meta.env.VITE_API_BASE_URL}/auth/refresh`,
            { refresh_token: refreshToken }
          );
          
          const { access_token, refresh_token } = response.data.data;
          login({
            user: useAuthStore.getState().user!,
            access_token,
            refresh_token,
          });
          
          originalRequest.headers.Authorization = `Bearer ${access_token}`;
          return client(originalRequest);
        }
      } catch (refreshError) {
        // 刷新 Token 失败，退出登录
        logout();
        window.location.href = '/login';
      }
    }
    
    // 处理其他错误
    const errorMessage = (error.response?.data as any)?.message || '请求失败';
    message.error(errorMessage);
    
    return Promise.reject(error);
  }
);

export default client;
```

## 6. 编辑器设计

### 6.1 Fabric.js 画布组件

```typescript
// src/components/editor/Canvas.tsx
import React, { useEffect, useRef } from 'react';
import { Canvas as FabricCanvas, FabricImage, IText } from 'fabric';
import { useEditorStore } from '../../stores/editorStore';
import { PageElement, ElementType } from '../../types/element';

interface CanvasProps {
  width: number;
  height: number;
}

const Canvas: React.FC<CanvasProps> = ({ width, height }) => {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const fabricRef = useRef<FabricCanvas | null>(null);
  
  const {
    setCanvas,
    setSelectedElement,
    currentPageId,
    pages,
    pushHistory,
  } = useEditorStore();
  
  // 初始化画布
  useEffect(() => {
    if (canvasRef.current && !fabricRef.current) {
      const canvas = new FabricCanvas(canvasRef.current, {
        width,
        height,
        backgroundColor: '#ffffff',
      });
      
      fabricRef.current = canvas;
      setCanvas(canvas);
      
      // 监听选择事件
      canvas.on('selection:created', (e) => {
        const selected = e.selected?.[0];
        if (selected) {
          setSelectedElement(selected as any);
        }
      });
      
      canvas.on('selection:updated', (e) => {
        const selected = e.selected?.[0];
        if (selected) {
          setSelectedElement(selected as any);
        }
      });
      
      canvas.on('selection:cleared', () => {
        setSelectedElement(null);
      });
      
      // 监听修改事件（用于撤销/重做）
      canvas.on('object:modified', () => {
        pushHistory(canvas.toJSON());
      });
    }
    
    return () => {
      fabricRef.current?.dispose();
      fabricRef.current = null;
    };
  }, [width, height]);
  
  // 加载页面内容
  useEffect(() => {
    const canvas = fabricRef.current;
    if (!canvas || !currentPageId) return;
    
    const currentPage = pages.find((p) => p.id === currentPageId);
    if (currentPage) {
      canvas.loadFromJSON(currentPage.canvas_data, () => {
        canvas.renderAll();
      });
    }
  }, [currentPageId, pages]);
  
  return (
    <canvas ref={canvasRef} className="border border-gray-300" />
  );
};

export default Canvas;
```

### 6.2 元素操作

```typescript
// src/hooks/useCanvas.ts
import { useCallback } from 'react';
import { useEditorStore } from '../stores/editorStore';
import { FabricImage, IText, FabricObject } from 'fabric';
import { ElementType, PageElement } from '../types/element';

export const useCanvas = () => {
  const {
    canvas,
    addElement,
    updateElement,
    deleteElement,
    selectedElement,
  } = useEditorStore();
  
  // 添加图片
  const addImage = useCallback(async (url: string, options?: Partial<PageElement>) => {
    if (!canvas) return;
    
    const img = await FabricImage.fromURL(url);
    img.set({
      left: options?.position_x || 100,
      top: options?.position_y || 100,
      width: options?.width || 200,
      height: options?.height || 150,
      ...options,
    });
    
    canvas.add(img);
    canvas.renderAll();
    
    // 保存到状态
    addElement({
      id: crypto.randomUUID(),
      element_type: ElementType.IMAGE,
      position_x: img.left || 0,
      position_y: img.top || 0,
      width: img.width || 0,
      height: img.height || 0,
      rotation: img.angle || 0,
      content: url,
      style_data: {},
    });
    
    return img;
  }, [canvas, addElement]);
  
  // 添加文字
  const addText = useCallback((text: string, options?: Partial<PageElement>) => {
    if (!canvas) return;
    
    const iText = new IText(text, {
      left: options?.position_x || 100,
      top: options?.position_y || 100,
      fontSize: options?.style_data?.fontSize || 24,
      fontFamily: options?.style_data?.fontFamily || 'Arial',
      fill: options?.style_data?.fill || '#000000',
      ...options,
    });
    
    canvas.add(iText);
    canvas.renderAll();
    
    // 保存到状态
    addElement({
      id: crypto.randomUUID(),
      element_type: ElementType.TEXT,
      position_x: iText.left || 0,
      position_y: iText.top || 0,
      width: iText.width || 0,
      height: iText.height || 0,
      rotation: iText.angle || 0,
      content: text,
      style_data: {
        fontSize: iText.fontSize,
        fontFamily: iText.fontFamily,
        fill: iText.fill,
      },
    });
    
    return iText;
  }, [canvas, addElement]);
  
  // 删除选中元素
  const deleteSelected = useCallback(() => {
    if (!canvas || !selectedElement) return;
    
    const activeObject = canvas.getActiveObject();
    if (activeObject) {
      canvas.remove(activeObject);
      canvas.renderAll();
      deleteElement(selectedElement.id);
    }
  }, [canvas, selectedElement, deleteElement]);
  
  // 撤销
  const undo = useCallback(() => {
    useEditorStore.getState().undo();
  }, []);
  
  // 重做
  const redo = useCallback(() => {
    useEditorStore.getState().redo();
  }, []);
  
  return {
    addImage,
    addText,
    deleteSelected,
    undo,
    redo,
  };
};
```

```typescript
// src/components/ai/GenerationProgress.tsx
import React, { useEffect, useState } from 'react';
import { Progress, Card, Tag, Button, Space } from 'antd';
import { CheckCircleOutlined, CloseCircleOutlined, SyncOutlined } from '@ant-design/icons';
import { TaskStatus, TaskResult } from '../../types/generation';

interface GenerationProgressProps {
  taskId: string;
  onComplete: (result: TaskResult) => void;
  onError: (error: string) => void;
}

const GenerationProgress: React.FC<GenerationProgressProps> = ({
  taskId,
  onComplete,
  onError,
}) => {
  const [task, setTask] = useState<TaskResult | null>(null);
  const [polling, setPolling] = useState(true);
  
  useEffect(() => {
    const pollTask = async () => {
      try {
        const response = await fetch(`/api/v1/generation/${taskId}`);
        const data = await response.json();
        setTask(data.data);
        
        if (data.data.status === TaskStatus.COMPLETED) {
          setPolling(false);
          onComplete(data.data);
        } else if (data.data.status === TaskStatus.FAILED) {
          setPolling(false);
          onError(data.data.error_message || '生成失败');
        }
      } catch (error) {
        console.error('轮询任务状态失败:', error);
      }
    };
    
    if (polling) {
      pollTask();
      const interval = setInterval(pollTask, 2000);
      return () => clearInterval(interval);
    }
  }, [taskId, polling, onComplete, onError]);
  
  const getStatusTag = () => {
    switch (task?.status) {
      case TaskStatus.PENDING:
        return <Tag icon={<SyncOutlined spin />} color="processing">等待中</Tag>;
      case TaskStatus.PROCESSING:
        return <Tag icon={<SyncOutlined spin />} color="processing">生成中</Tag>;
      case TaskStatus.COMPLETED:
        return <Tag icon={<CheckCircleOutlined />} color="success">已完成</Tag>;
      case TaskStatus.FAILED:
        return <Tag icon={<CloseCircleOutlined />} color="error">失败</Tag>;
      default:
        return null;
    }
  };
  
  const getProgress = () => {
    switch (task?.status) {
      case TaskStatus.PENDING:
        return 10;
      case TaskStatus.PROCESSING:
        return 50;
      case TaskStatus.COMPLETED:
        return 100;
      case TaskStatus.FAILED:
        return 100;
      default:
        return 0;
    }
  };
  
  return (
    <Card title="生成进度">
      <Space direction="vertical" style={{ width: '100%' }}>
        <div>{getStatusTag()}</div>
        <Progress
          percent={getProgress()}
          status={task?.status === TaskStatus.FAILED ? 'exception' : 'active'}
        />
        {task?.status === TaskStatus.COMPLETED && task.result && (
          <div className="generation-result">
            <img
              src={task.result.image_url}
              alt="生成结果"
              style={{ maxWidth: '100%', borderRadius: '8px' }}
            />
          </div>
        )}
      </Space>
    </Card>
  );
};

export default GenerationProgress;
```

## 10. 错误处理

### 10.1 全局错误边界

```typescript
// src/components/common/ErrorBoundary.tsx
import React, { Component, ErrorInfo, ReactNode } from 'react';
import { Result, Button } from 'antd';

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false, error: null };
  }
  
  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }
  
  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('组件错误:', error, errorInfo);
  }
  
  handleRetry = () => {
    this.setState({ hasError: false, error: null });
  };
  
  render() {
    if (this.state.hasError) {
      return (
        <Result
          status="error"
          title="页面出错了"
          subTitle={this.state.error?.message || '未知错误'}
          extra={
            <Button type="primary" onClick={this.handleRetry}>
              重试
            </Button>
          }
        />
      );
    }
    
    return this.props.children;
  }
}

export default ErrorBoundary;
```
