## 1. 概述

### 1.1 存储需求

系统需要存储以下类型的文件：

| 文件类型 | 说明 | 大小范围 | 访问频率 |
|---------|------|---------|---------|
| 原始生成图片 | AI 生成的原始图片 | 1-10MB | 低 |
| 编辑后图片 | 用户编辑保存的图片 | 1-10MB | 中 |
| 用户上传图片 | 用户自行上传的图片 | 0.1-10MB | 中 |
| 缩略图 | 页面预览缩略图 | 10-100KB | 高 |
| 导出文件 | PDF/PNG/JPEG 导出 | 1-50MB | 低 |
| 用户头像 | 用户头像图片 | 10-500KB | 高 |

### 1.2 设计原则

1. **可扩展性**: 支持从本地存储迁移到对象存储
2. **成本效益**: 课程项目使用本地存储，生产环境使用对象存储
3. **安全性**: 文件类型检查、大小限制、路径安全
4. **性能**: 缩略图加速访问，CDN 加速（可选）

```cpp
std::string FileUtils::generatePath(

    const std::string &userId,
    const std::string &bookId,

    const std::string &fileName

    path << basePath << "/users/books/" << userId << "/" << bookId << "/" << subDir << "/" << fileName;

    const std::string &basePath,
    const std::string &userId,
    const std::string &extension

    std::ostringstream path;

    return path.str();
}

std::string FileUtils::getExtension(const std::string &fileName) {
    size_t dotPos = fileName.rfind('.');
    if (dotPos != std::string::npos) {
        return fileName.substr(dotPos + 1);
    }
    return "";
}

bool FileUtils::isAllowedType(const std::string &fileName, const std::vector<std::string> &allowedTypes) {
    std::string ext = getExtension(fileName);
    std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
    
    for (const auto &type : allowedTypes) {
        if (ext == type) {
            return true;
        }
    }
    return false;
}

bool FileUtils::createDirectory(const std::string &path) {
    try {
        return fs::create_directories(path);
    } catch (const std::exception &) {
        return false;
    }
}

bool FileUtils::deleteFile(const std::string &filePath) {
    try {
        return fs::remove(filePath);
    } catch (const std::exception &) {
        return false;
    }
}
```

```cpp
// utils/MinioClient.h
#pragma once
#include <string>
#include <vector>

class MinioClient {
public:
    MinioClient(
        const std::string &endpoint,
        const std::string &accessKey,
        const std::string &secretKey,
        const std::string &bucket
    );
    
    // 上传文件
    bool uploadFile(const std::string &objectName, const std::string &filePath);
    
    // 下载文件
    bool downloadFile(const std::string &objectName, const std::string &filePath);
    
    // 删除文件
    bool deleteFile(const std::string &objectName);
    
    // 获取文件 URL
    std::string getFileUrl(const std::string &objectName, int expiry = 3600);
    
    // 列出文件
    std::vector<std::string> listFiles(const std::string &prefix);
    
    // 检查文件是否存在
    bool fileExists(const std::string &objectName);

private:
    std::string endpoint_;
    std::string accessKey_;
    std::string secretKey_;
    std::string bucket_;
};

// utils/MinioClient.cc
#include "MinioClient.h"
#include <curl/curl.h>
#include <openssl/hmac.h>
#include <openssl/sha.h>

MinioClient::MinioClient(
    const std::string &endpoint,
    const std::string &accessKey,
    const std::string &secretKey,
    const std::string &bucket
) : endpoint_(endpoint)
  , accessKey_(accessKey)
  , secretKey_(secretKey)
  , bucket_(bucket) {
    curl_global_init(CURL_GLOBAL_ALL);
}

bool MinioClient::uploadFile(const std::string &objectName, const std::string &filePath) {
    CURL *curl = curl_easy_init();
    if (!curl) return false;
    
    std::string url = endpoint_ + "/" + bucket_ + "/" + objectName;
    
    FILE *file = fopen(filePath.c_str(), "rb");
    if (!file) {
        curl_easy_cleanup(curl);
        return false;
    }
    
    curl_easy_setopt(curl, CURLOPT_URL, url.c_str());
    curl_easy_setopt(curl, CURLOPT_UPLOAD, 1L);
    curl_easy_setopt(curl, CURLOPT_READDATA, file);
    
    // 设置认证头
    std::string authHeader = generateAuthHeader("PUT", objectName);
    struct curl_slist *headers = nullptr;
    headers = curl_slist_append(headers, authHeader.c_str());
    curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
    
    CURLcode res = curl_easy_perform(curl);
    
    fclose(file);
    curl_slist_free_all(headers);
    curl_easy_cleanup(curl);
    
    return (res == CURLE_OK);
}

std::string MinioClient::getFileUrl(const std::string &objectName, int expiry) {
    // 生成预签名 URL
    time_t expires = time(nullptr) + expiry;
    std::string stringToSign = "GET\n\n\n" + std::to_string(expires) + "\n/" + bucket_ + "/" + objectName;
    
    unsigned char digest[SHA256_DIGEST_LENGTH];
    HMAC(EVP_sha256(), secretKey_.c_str(), secretKey_.length(),
         (unsigned char*)stringToSign.c_str(), stringToSign.length(),
         digest, NULL);
    
    // Base64 编码
    std::string signature = base64Encode(digest, SHA256_DIGEST_LENGTH);
    
    return endpoint_ + "/" + bucket_ + "/" + objectName +
           "?Expires=" + std::to_string(expires) +
           "&AWSAccessKeyId=" + accessKey_ +
           "&Signature=" + urlEncode(signature);
}
```

```cpp
bool LocalStorageProvider::uploadFile(const std::string &objectName, const std::string &filePath) {
    std::string destPath = basePath_ + "/" + objectName;
    
    // 创建目标目录
    size_t lastSlash = destPath.rfind('/');
    if (lastSlash != std::string::npos) {
        FileUtils::createDirectory(destPath.substr(0, lastSlash));
    }
    
    try {
        fs::copy_file(filePath, destPath, fs::copy_options::overwrite_existing);
        return true;
    } catch (const std::exception &) {
        return false;
    }
}

    // 本地存储返回文件路径（需要通过静态文件服务访问）
    return "/storage/" + objectName;
```

```cpp
bool ImageProcessor::createThumbnail(
    const std::string &inputPath,
    const std::string &outputPath,
    int width,
    int height
) {
    cv::Mat image = cv::imread(inputPath);
    if (image.empty()) return false;
    
    cv::Mat thumbnail;
    cv::resize(image, thumbnail, cv::Size(width, height));
    
    return cv::imwrite(outputPath, thumbnail);
}

bool ImageProcessor::resizeImage(
    const std::string &inputPath,
    const std::string &outputPath,
    int width,
    int height
) {
    cv::Mat image = cv::imread(inputPath);
    if (image.empty()) return false;
    
    cv::Mat resized;
    cv::resize(image, resized, cv::Size(width, height));
    
    return cv::imwrite(outputPath, resized);
}

ImageProcessor::ImageInfo ImageProcessor::getImageInfo(const std::string &filePath) {
    ImageInfo info;
    
    cv::Mat image = cv::imread(filePath);
    if (!image.empty()) {
        info.width = image.cols;
        info.height = image.rows;
    }
    
    // 获取文件大小
    std::ifstream file(filePath, std::ios::binary | std::ios::ate);
    info.fileSize = file.tellg();
    
    // 获取格式
    size_t dotPos = filePath.rfind('.');
    if (dotPos != std::string::npos) {
        info.format = filePath.substr(dotPos + 1);
    }
    
    return info;
}
```

## 6. 文件上传安全

### 6.1 文件类型检查

```cpp
// utils/FileValidator.h
#pragma once
#include <string>
#include <vector>

class FileValidator {
public:
    // 允许的图片类型
    static const std::vector<std::string> ALLOWED_IMAGE_TYPES;
    
    // 允许的文档类型
    static const std::vector<std::string> ALLOWED_DOCUMENT_TYPES;
    
    // 验证文件类型
    static bool isValidFileType(const std::string &fileName, const std::vector<std::string> &allowedTypes);
    
    // 验证文件大小
    static bool isValidFileSize(size_t fileSize, size_t maxSize);
    
    // 验证文件内容（检查魔数）
    static bool isValidFileContent(const std::string &filePath, const std::string &expectedType);
    
    // 检查文件是否包含恶意内容
    static bool isSafeFile(const std::string &filePath);
};

// utils/FileValidator.cc
#include "FileValidator.h"
#include <fstream>
#include <algorithm>

const std::vector<std::string> FileValidator::ALLOWED_IMAGE_TYPES = {
    "jpg", "jpeg", "png", "gif", "webp"
};

const std::vector<std::string> FileValidator::ALLOWED_DOCUMENT_TYPES = {
    "pdf"
};

bool FileValidator::isValidFileType(const std::string &fileName, const std::vector<std::string> &allowedTypes) {
    size_t dotPos = fileName.rfind('.');
    if (dotPos == std::string::npos) return false;
    
    std::string ext = fileName.substr(dotPos + 1);
    std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
    
    return std::find(allowedTypes.begin(), allowedTypes.end(), ext) != allowedTypes.end();
}

bool FileValidator::isValidFileContent(const std::string &filePath, const std::string &expectedType) {
    std::ifstream file(filePath, std::ios::binary);
    if (!file) return false;
    
    // 读取文件头（魔数）
    unsigned char header[8];
    file.read(reinterpret_cast<char*>(header), 8);
    
    // JPEG: FF D8 FF
    if (expectedType == "jpg" || expectedType == "jpeg") {
        return header[0] == 0xFF && header[1] == 0xD8 && header[2] == 0xFF;
    }
    
    // PNG: 89 50 4E 47 0D 0A 1A 0A
    if (expectedType == "png") {
        return header[0] == 0x89 && header[1] == 0x50 && 
               header[2] == 0x4E && header[3] == 0x47;
    }
    
    // GIF: 47 49 46 38
    if (expectedType == "gif") {
        return header[0] == 0x47 && header[1] == 0x49 && 
               header[2] == 0x46 && header[3] == 0x38;
    }
    
    return true;
}
```

```cpp
    // 清理未使用的图片
    static void cleanupUnusedImages();
};

// services/CleanupService.cc
#include "CleanupService.h"
#include <filesystem>
#include <chrono>

namespace fs = std::filesystem;

void CleanupService::cleanupTempFiles(int maxAgeHours) {
    std::string tempPath = "./storage/temp";
    
    auto now = std::chrono::system_clock::now();
    
    for (const auto &entry : fs::directory_iterator(tempPath)) {
        if (fs::is_directory(entry)) {
            auto lastWriteTime = fs::last_write_time(entry);
            auto age = std::chrono::duration_cast<std::chrono::hours>(
                now - lastWriteTime
            );
            
            if (age.count() > maxAgeHours) {
                fs::remove_all(entry);
            }
        }
    }
}
```

### 8.2 文件哈希去重

```cpp
// 计算文件哈希
std::string calculateFileHash(const std::string &filePath) {
    std::ifstream file(filePath, std::ios::binary);
    SHA256_CTX sha256;
    SHA256_Init(&sha256);
    
    char buffer[8192];
    while (file.read(buffer, sizeof(buffer))) {
        SHA256_Update(&sha256, buffer, file.gcount());
    }
    
    unsigned char hash[SHA256_DIGEST_LENGTH];
    SHA256_Final(hash, &sha256);
    
    // 转换为十六进制字符串
    std::stringstream ss;
    for (int i = 0; i < SHA256_DIGEST_LENGTH; i++) {
        ss << std::hex << std::setw(2) << std::setfill('0') << (int)hash[i];
    }
    
    return ss.str();
}
```

## 10. 迁移策略

### 10.1 从本地存储迁移到 MinIO

```cpp
// scripts/migrate_to_minio.cpp
#include "utils/MinioClient.h"
#include "utils/FileUtils.h"

void migrateToMinio(const std::string &localPath, MinioClient &minioClient) {
    for (const auto &entry : fs::recursive_directory_iterator(localPath)) {
        if (fs::is_regular_file(entry)) {
            std::string relativePath = entry.path().string().substr(localPath.length() + 1);
            
            // 上传到 MinIO
            if (minioClient.uploadFile(relativePath, entry.path().string())) {
                std::cout << "已迁移: " << relativePath << std::endl;
                
                // 可选：删除本地文件
                // fs::remove(entry);
            } else {
                std::cerr << "迁移失败: " << relativePath << std::endl;
            }
        }
    }
}
```
