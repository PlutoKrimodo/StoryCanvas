```cpp
// utils/PasswordUtils.h
#pragma once
#include <string>

class PasswordUtils {
public:
    // 生成密码哈希（使用 bcrypt）
    static std::string hashPassword(const std::string &password);
    
    // 验证密码
    static bool verifyPassword(const std::string &password, const std::string &hash);
    
    // 生成盐值
    static std::string generateSalt();
    
    // 密码强度检查
    static bool isStrongPassword(const std::string &password);

private:
    static const int BCRYPT_COST = 12;
};

// utils/PasswordUtils.cc
#include "PasswordUtils.h"
#include <bcrypt/bcrypt.h>
#include <regex>

std::string PasswordUtils::hashPassword(const std::string &password) {
    char salt[BCRYPT_HASHSIZE];
    char hash[BCRYPT_HASHSIZE];
    
    // 生成盐值
    bcrypt_gensalt(BCRYPT_COST, salt);
    
    // 生成哈希
    bcrypt_hashpw(password.c_str(), salt, hash);
    
    return std::string(hash);
}

bool PasswordUtils::verifyPassword(const std::string &password, const std::string &hash) {
    return bcrypt_checkpw(password.c_str(), hash.c_str()) == 0;
}

bool PasswordUtils::isStrongPassword(const std::string &password) {
    // 至少 8 位
    if (password.length() < 8) return false;
    
    // 包含大写字母
    if (!std::regex_search(password, std::regex("[A-Z]"))) return false;
    
    // 包含小写字母
    if (!std::regex_search(password, std::regex("[a-z]"))) return false;
    
    // 包含数字
    if (!std::regex_search(password, std::regex("[0-9]"))) return false;
    
    // 包含特殊字符
    if (!std::regex_search(password, std::regex("[!@#$%^&*(),.?\":{}|<>]"))) return false;
    
    return true;
}
```

> **实现说明**：后端优先使用 `libbcrypt`（上例 API）；若编译环境未安装该库，则自动回退为
> **PBKDF2-HMAC-SHA256**（OpenSSL 实现，迭代 12 万次）。哈希串自描述，校验时自动识别算法。

```cpp
// utils/JwtUtils.h
#pragma once
#include <string>
#include <optional>
#include <jwt-cpp/jwt.h>

struct TokenPayload {
    std::string userId;
    std::string username;
    time_t issuedAt;
    time_t expiresAt;
};

struct TokenPair {
    std::string accessToken;
    std::string refreshToken;
};

class JwtUtils {
public:
    // 生成 Token 对
    static TokenPair generateTokenPair(
        const std::string &userId,
        const std::string &username
    );
    
    // 验证 Token
    static std::optional<TokenPayload> verifyToken(const std::string &token);
    
    // 刷新 Token
    static std::optional<std::string> refreshToken(const std::string &refreshToken);
    
    // 撤销 Token
    static void revokeToken(const std::string &token);

private:
    static std::string secret_;
    static int accessTokenExpiry_;  // 1 小时
    static int refreshTokenExpiry_; // 7 天
};

// utils/JwtUtils.cc
#include "JwtUtils.h"
#include <chrono>

std::string JwtUtils::secret_ = "your-secret-key";
int JwtUtils::accessTokenExpiry_ = 3600;
int JwtUtils::refreshTokenExpiry_ = 604800;

TokenPair JwtUtils::generateTokenPair(
    const std::string &userId,
    const std::string &username
) {
    auto now = std::chrono::system_clock::now();
    
    // 生成 Access Token
    auto accessToken = jwt::create()
        .set_type("JWT")
        .set_issuer("storycanvas")
        .set_subject(userId)
        .set_payload_claim("username", jwt::claim(username))
        .set_issued_at(now)
        .set_expires_at(now + std::chrono::seconds(accessTokenExpiry_))
        .sign(jwt::algorithm::hs256{secret_});
    
    // 生成 Refresh Token
    auto refreshToken = jwt::create()
        .set_type("JWT")
        .set_issuer("storycanvas")
        .set_subject(userId)
        .set_payload_claim("username", jwt::claim(username))
        .set_issued_at(now)
        .set_expires_at(now + std::chrono::seconds(refreshTokenExpiry_))
        .sign(jwt::algorithm::hs256{secret_});
    
    return {accessToken, refreshToken};
}

std::optional<TokenPayload> JwtUtils::verifyToken(const std::string &token) {
    try {
        auto decoded = jwt::decode()
            .verify(jwt::algorithm::hs256{secret_})
            .decode(token);
        
        TokenPayload payload;
        payload.userId = decoded.get_subject();
        payload.username = decoded.get_payload_claim("username").as_string();
        payload.issuedAt = decoded.get_issued_at();
        payload.expiresAt = decoded.get_expires_at();
        
        // 检查是否已过期
        auto now = std::chrono::system_clock::now();
        if (std::chrono::system_clock::now().time_since_epoch().count() > payload.expiresAt) {
            return std::nullopt;
        }
        
        // 检查是否已撤销
        if (isTokenRevoked(token)) {
            return std::nullopt;
        }
        
        return payload;
    } catch (const std::exception &) {
        return std::nullopt;
    }
}

void JwtUtils::revokeToken(const std::string &token) {
    // 将 Token 添加到黑名单（可以使用 Redis）
    // 这里简化为内存存储
    static std::set<std::string> revokedTokens;
    revokedTokens.insert(token);
}
```

```cpp
#pragma once
#include <drogon/HttpFilter.h>

class AuthorizationFilter : public drogon::HttpFilter<AuthorizationFilter> {
public:
    void doFilter(const HttpRequestPtr &req,
                  FilterCallback &&fcb,
                  FilterChainCallback &&fccb) override;
    
    // 检查用户权限
    bool checkPermission(const std::string &userId, const std::string &resource, const std::string &action);
    
    // 检查资源所有权
    bool checkOwnership(const std::string &userId, const std::string &resourceId, const std::string &resourceType);
};

// filters/AuthorizationFilter.cc
#include "AuthorizationFilter.h"
#include "../repositories/UserRepository.h"
#include "../repositories/BookRepository.h"

void AuthorizationFilter::doFilter(const HttpRequestPtr &req,
                                   FilterCallback &&fcb,
                                   FilterChainCallback &&fccb) {
    // 获取用户 ID
    auto userId = req->getAttributes<std::string>("user_id");
    if (userId.empty()) {
        auto resp = HttpResponse::newHttpResponse(k401Unauthorized, CT_APPLICATION_JSON);
        resp->setBody("{\"code\":401,\"message\":\"未授权\"}");
        fcb(resp);
        return;
    }
    
    // 获取请求的资源
    auto path = req->getPath();
    auto method = req->getMethod();
    
    // 检查权限
    if (!checkPermission(userId, path, methodToString(method))) {
        auto resp = HttpResponse::newHttpResponse(k403Forbidden, CT_APPLICATION_JSON);
        resp->setBody("{\"code\":403,\"message\":\"权限不足\"}");
        fcb(resp);
        return;
    }
    
    // 继续处理
    fccb();
}

bool AuthorizationFilter::checkPermission(const std::string &userId, const std::string &resource, const std::string &action) {
    // 检查用户是否有权限访问资源
    // 这里可以根据实际需求实现更复杂的权限逻辑
    
    // 简单示例：用户只能访问自己的资源
    if (resource.find("/books/") != std::string::npos) {
        // 提取 bookId
        size_t pos = resource.find("/books/") + 7;
        size_t end = resource.find("/", pos);
        std::string bookId = resource.substr(pos, end - pos);
        
        return checkOwnership(userId, bookId, "book");
    }
    
    return true;
}

bool AuthorizationFilter::checkOwnership(const std::string &userId, const std::string &resourceId, const std::string &resourceType) {
    if (resourceType == "book") {
        BookRepository bookRepo;
        auto book = bookRepo.findById(resourceId);
        return book && book->userId == userId;
    }
    
    return false;
}
```

## 4. API 安全

### 4.1 速率限制

**已落地（2026-10-15，Q5）**：内存固定窗口限流，规格对齐 `docs/16` §7.4。

实现位置：`backend/filters/RateLimitFilter.cpp`（策略）+ `backend/utils/RateLimiter.cpp`（计数），
在 `main.cpp` 以**同步 advice** 全局注册——Drogon 1.9 的「全局过滤器」需走 pre-routing
advice 形式（libdrogon 的 GlobalFilters 插件即如此实现），故该类不继承 `HttpFilter`，
但沿用过滤器职责划分（所有请求先过限流，再进路由与路由级过滤器）。

| 档位 | 键 | 缺省额度 | 说明 |
|------|-----|---------|------|
| 匿名 | 客户端 IP | 100 次/分 | 无有效 Access Token（缺失 / 非法 / 过期均按匿名计） |
| 认证 | 用户 ID | 1000 次/分 | 有效 Access Token（正式用户与游客同档） |
| AI 生成 | 用户 ID / IP | 10 次/分 | 任务创建与重新生成（都会触发真实计费调用） |
| 游客会话签发 | 客户端 IP | 10 次/分 | `POST /auth/guest`，防无限换取游客身份（docs/18 Q3①） |

实现要点与取舍：

- **固定窗口（60s）**：实现简单、无 Redis 依赖；窗口边界处允许瞬时两倍流量，属 MVP 可接受取舍。计数保存在进程内存（`mutex` 保护，周期清扫过期键防映射膨胀），重启清零；键空间已按维度隔离（`ip:` / `user:` / `generation:` …），后续可平滑迁移 Redis；
- **身份识别**：限流先于路由执行，直接校验 `Authorization` 中的 Access Token（`JwtUtils::verifyToken`）：有效则按用户分桶，否则按 IP 分桶；
- **客户端 IP**：直连取对端地址；对端为回环 / 内网地址（本机 Vite 代理、nginx / Docker 网关）时优先采用 `X-Real-IP`（nginx.conf 已透传）。来自内网的对端可伪造该头绕过 IP 档位，属已知取舍——成本封顶仍由「全站每日额度」兜底；
- **豁免**：`OPTIONS` 预检、`/health` 与 `/api/v1/health`、生成图 `GET .../raw`（`<img>` 标签直连加载、无 Authorization 头，且为只读资源）；
- **响应**：`429` + 统一 `{code,message,data:null}` 结构 + `Retry-After: 60`，并记录 WARN 日志（rule / key，不含凭据）；
- **配置**：`RATE_LIMIT_ENABLED` / `RATE_LIMIT_ANON_PER_MIN` / `RATE_LIMIT_AUTH_PER_MIN` / `RATE_LIMIT_GENERATION_PER_MIN` / `RATE_LIMIT_GUEST_TOKEN_PER_MIN`（见 `.env.example`；`<=0` 表示该档位不限制，仅建议开发 / 压测时关闭总开关）；
- **验收**：`scripts/verify_security.sh`（匿名超限、游客签发超限、认证不受匿名额度影响）；单元测试 `backend/tests/test_rate_limiter.cpp`。

配套的**游客写接口拦截**（Q3②）见 `backend/filters/MemberFilter.cpp`：挂在 `JwtFilter`
之后（依赖其写入的 `is_guest` 属性），对「创建 / 更新 / 删除绘本、保存页」等写接口
返回 403，游客只读不写（§3.2 权限控制的落地）。

```cpp
std::string Validator::escapeHtml(const std::string &input) {
    std::string output = input;
    
    // 替换 HTML 特殊字符
    std::vector<std::pair<std::string, std::string>> replacements = {
        {"&", "&amp;"},
        {"<", "&lt;"},
        {">", "&gt;"},
        {"\"", "&quot;"},
        {"'", "&#x27;"}
    };
    
    for (const auto &[from, to] : replacements) {
        size_t pos = 0;
        while ((pos = output.find(from, pos)) != std::string::npos) {
            output.replace(pos, from.length(), to);
            pos += to.length();
        }
    }
    
    return output;
}
```

```cpp
// utils/FileValidator.h
#pragma once
#include <string>
#include <vector>

class FileValidator {
public:
    // 允许的文件类型
    static const std::vector<std::string> ALLOWED_IMAGE_TYPES;
    static const std::vector<std::string> ALLOWED_DOCUMENT_TYPES;
    
    // 验证文件类型
    static bool isValidFileType(const std::string &fileName, const std::vector<std::string> &allowedTypes);
    
    // 验证文件大小
    static bool isValidFileSize(size_t fileSize, size_t maxSize);
    
    // 验证文件内容（检查魔数）
    static bool isValidFileContent(const std::string &filePath, const std::string &expectedType);
    
    // 检查文件是否包含恶意内容
    static bool isSafeFile(const std::string &filePath);
    
    // 防止路径穿越
    static bool isPathSafe(const std::string &path);
};

// utils/FileValidator.cc
#include "FileValidator.h"
#include <fstream>
#include <algorithm>
#include <filesystem>

const std::vector<std::string> FileValidator::ALLOWED_IMAGE_TYPES = {
    "jpg", "jpeg", "png", "gif", "webp"
};

const std::vector<std::string> FileValidator::ALLOWED_DOCUMENT_TYPES = {
    "pdf"
};

bool FileValidator::isValidFileType(const std::string &fileName, const std::vector<std::string> &allowedTypes) {
    size_t dotPos = fileName.rfind('.');
    if (dotPos == std::string::npos) return false;
    
    std::string ext = fileName.substr(dotPos + 1);
    std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
    
    return std::find(allowedTypes.begin(), allowedTypes.end(), ext) != allowedTypes.end();
}

bool FileValidator::isValidFileContent(const std::string &filePath, const std::string &expectedType) {
    std::ifstream file(filePath, std::ios::binary);
    if (!file) return false;
    
    // 读取文件头（魔数）
    unsigned char header[8];
    file.read(reinterpret_cast<char*>(header), 8);
    
    // JPEG: FF D8 FF
    if (expectedType == "jpg" || expectedType == "jpeg") {
        return header[0] == 0xFF && header[1] == 0xD8 && header[2] == 0xFF;
    }
    
    // PNG: 89 50 4E 47 0D 0A 1A 0A
    if (expectedType == "png") {
        return header[0] == 0x89 && header[1] == 0x50 && 
               header[2] == 0x4E && header[3] == 0x47;
    }
    
    return true;
}

bool FileValidator::isPathSafe(const std::string &path) {
    // 检查路径是否包含 .. 或其他危险字符
    if (path.find("..") != std::string::npos) {
        return false;
    }
    
    // 检查是否是绝对路径
    if (path[0] == '/') {
        return false;
    }
    
    // 检查是否包含特殊字符
    std::vector<char> dangerousChars = {'<', '>', '|', '*', '?', '"'};
    for (char c : dangerousChars) {
        if (path.find(c) != std::string::npos) {
            return false;
        }
    }
    
    return true;
}
```

```cpp
ServiceResult ImageService::uploadImage(const UploadImageRequest &request) {
    ServiceResult result;
    
    // 1. 验证文件类型
    if (!FileValidator::isValidFileType(request.fileName, FileValidator::ALLOWED_IMAGE_TYPES)) {

        result.success = false;
        result.message = "文件内容无效";

    return result;
```

## 7. Prompt 安全

### 7.1 输入验证

```cpp
// utils/PromptValidator.h
#pragma once
#include <string>

class PromptValidator {
public:
    // 验证 Prompt
    static bool isValidPrompt(const std::string &prompt);
    
    // 清理 Prompt
    static std::string sanitizePrompt(const std::string &prompt);
    
    // 检查是否包含敏感内容
    static bool containsSensitiveContent(const std::string &prompt);
    
    // 限制 Prompt 长度
    static std::string truncatePrompt(const std::string &prompt, size_t maxLength);
};

// utils/PromptValidator.cc
#include "PromptValidator.h"
#include <algorithm>

bool PromptValidator::isValidPrompt(const std::string &prompt) {
    // 检查长度
    if (prompt.empty() || prompt.length() > 2000) {
        return false;
    }
    
    // 检查是否包含恶意内容
    if (containsSensitiveContent(prompt)) {
        return false;
    }
    
    return true;
}

std::string PromptValidator::sanitizePrompt(const std::string &prompt) {
    std::string sanitized = prompt;
    
    // 移除危险字符
    sanitized.erase(std::remove_if(sanitized.begin(), sanitized.end(), [](char c) {
        return c == '<' || c == '>' || c == '{' || c == '}' || c == '"';
    }), sanitized.end());
    
    // 截断过长的 Prompt
    if (sanitized.length() > 2000) {
        sanitized = sanitized.substr(0, 2000);
    }
    
    return sanitized;
}

bool PromptValidator::containsSensitiveContent(const std::string &prompt) {
    // 检查敏感词（示例）
    std::vector<std::string> sensitiveWords = {
        "violence", "gore", "nude", "porn", "illegal"
    };
    
    std::string lowerPrompt = prompt;
    std::transform(lowerPrompt.begin(), lowerPrompt.end(), lowerPrompt.begin(), ::tolower);
    
    for (const auto &word : sensitiveWords) {
        if (lowerPrompt.find(word) != std::string::npos) {
            return true;
        }
    }
    
    return false;
}
```

### 8.4 密钥存放与安全 push 流程

**存放约定（唯一的写入口）**

| 凭据 | 位置 | 版本控制状态 |
|---|---|---|
| DB / JWT / AI_SERVICE_API_KEY | 根目录 `.env` | 未跟踪，`.gitignore` 的 `.env` 规则对任意层级生效 |
| 厂商 key、INTERNAL_API_KEY | `ai-service/.env` | 未跟踪 |
| key 的**名称**与占位值 | `.env.example` / `ai-service/.env.example` | 跟踪，值一律写 `change_me_*` |

真实 key 只写进 `.env` 文件本身。**不要把 key 贴进 shell 命令行的参数、也不要贴进聊天记录**——执行记录会落进 `.qoder/settings.local.json`，这类"命令记录文件"正是历史上那次密钥入库的实际通道，该文件已取消跟踪并写入 `.gitignore`。

**本地关卡：`.githooks/pre-commit` + `.githooks/pre-push`**

零依赖（只需 `git` + `grep` + `awk` + `sed`），共用 `.githooks/secret-scan.sh`。服务端检查不可作为唯一防线：GitHub 的 push protection 是**事后**拦截，另一远端没有等价检查——所以拦截必须发生在 `git push` 之前。

- 关卡 A 文件名：`.env`、`*.pem`、`*.key`、`credentials`、`settings.local.json` 等出现即拒绝（`.env.example` 类模板除外）
- 关卡 B 内容：`sk-` 长串、`*_API_KEY/*_SECRET/*_TOKEN/*_PASSWORD = 长值`、`Authorization: Bearer …`、`VITE_*KEY=…`（前端 env 会被打进公开 bundle，属于最贵的一类误提交）
- 关卡 C 白名单：`change_me*`、`your_*`、`<…>`、`xxxxxxxx`、`${VAR}`、空值 → 放行，模板文件不会误报
- 只扫描**新增行**，因此历史里已存在的内容不会让每次提交都失败
- 命中只打印 `文件:行号 + 脱敏后的变量名`，**永不回显完整值**
- 单行豁免 `# allow-secret`；整体豁免 `SKIP_SECRET_CHECK=1`（会显式输出"本次未被检查"）

安装（`scripts/setup.sh` 已包含，hook 不随 clone 生效）：

```bash
git config core.hooksPath .githooks
chmod +x .githooks/*
sh .githooks/pre-commit --report-tracked   # 全量校准，只打印不拦截
```

**填真实 key 之后的 push 流程**

1. 用编辑器把 key 写进 `.env` / `ai-service/.env`；随后 `git diff --stat` 应显示跟踪文件无变化
2. `git status --short` 逐项确认；断言下列命令**必须空输出**：
   `git ls-files | grep -E '(^|/)\.env$|\.qoder/'`
3. 只 `git add <明确路径>`，不用 `git add -A` / `git add .`
4. `git commit` → pre-commit 拦截；有命中先看 `git diff --cached -- <文件>`，不要急着加豁免
5. `git push --dry-run origin main` 先走一遍 pre-push（无网络副作用），通过后再正式推送各远端
6. 已 commit 但**未 push**：`git reset --soft HEAD~1` 重做
7. **已经 push**：顺序永远是「先在厂商控制台吊销/删除 key → 再改写历史 → 最后重推」；不要使用平台给出的"放行该密钥"链接
8. 备份镜像（如 `/tmp` 下的 mirror 仓库）含改写前历史，用完即删

**关卡边界（必须知道的三点）**

- 关卡只看**新增行**：历史里已经存在的值不会再次拦截。存量用 `sh .githooks/pre-commit --report-tracked` 查看（只打印、不拦截、不改动）
- `# allow-secret` 与 `SKIP_SECRET_CHECK=1` 是逃生阀，绕过 pre-commit 之后，最后一道门是 pre-push——它扫的是「远端还没有的那段提交」，两个远端走同一道门
- 现存残留：开发文档里留有一个已入库的数据库口令（历史已公开，具体位置用 `--report-tracked` 自查）。正确处置是**改数据库口令并同步 `.env`**，而不是改写文档历史；本仓库按演示成本选择不做历史重写

## 12. 总结

### 12.1 课程项目安全重点

1. **密码安全**: 使用 bcrypt 加盐哈希（cost ≥ 12），禁止明文/可逆加密
2. **加密存储**: 密钥类信息不落库（环境变量注入）；确需落库的敏感字段用 AES-256-GCM
3. **JWT 认证**: 正确实现 Token 管理
4. **输入验证**: 防止注入攻击
5. **文件安全**: 类型和大小检查
6. **资源隔离**: 用户只能访问自己的数据
7. **日志脱敏**: 密码、Token、密钥不落日志

### 12.2 可以简化的部分

1. 双因素认证（可选）
2. 高级入侵检测（可选）
