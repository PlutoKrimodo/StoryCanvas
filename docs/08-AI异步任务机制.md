## 1. 概述

### 1.1 为什么需要异步任务

AI 图像生成通常需要较长时间（10-60秒），如果采用同步方式：
- HTTP 请求会一直阻塞
- 用户体验差
- 服务器资源浪费
- 无法处理并发请求

### 1.2 异步任务流程

```
用户请求 → 创建任务 → 返回任务 ID → 后台执行 → 更新状态 → 用户轮询获取结果
```

## 2. 任务状态设计

### 2.1 状态枚举

```
┌─────────┐     ┌─────────────┐     ┌─────────────┐
│ PENDING │────▶│ PROCESSING  │────▶│  COMPLETED  │
└─────────┘     └─────────────┘     └─────────────┘
                       │                   │
                       │                   │
                       ▼                   ▼
                 ┌─────────┐        ┌─────────┐
                 │  FAILED │        │ CANCELLED│
                 └─────────┘        └─────────┘
```

### 2.2 状态说明

| 状态 | 说明 | 可转换状态 |
|------|------|-----------|
| PENDING | 待处理，等待执行 | PROCESSING, CANCELLED |
| PROCESSING | 处理中，正在生成 | COMPLETED, FAILED |
| COMPLETED | 已完成，生成成功 | - |
| FAILED | 失败，生成失败 | - |
| CANCELLED | 已取消 | - |

#### 3.1.1 C++ 后端实现

```cpp
// services/GenerationService.h
#pragma once
#include <queue>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <unordered_map>
#include "../models/GenerationTask.h"
#include "../repositories/GenerationTaskRepository.h"

class GenerationService {
public:
    GenerationService();
    ~GenerationService();
    
    // 创建生成任务
    std::string createTask(const GenerationRequest &request);
    
    // 获取任务状态
    std::optional<GenerationTask> getTaskStatus(const std::string &taskId);
    
    // 取消任务
    bool cancelTask(const std::string &taskId);
    
    // 启动工作线程
    void startWorker();
    
    // 停止工作线程
    void stopWorker();

private:
    // 任务队列
    std::queue<std::string> taskQueue_;
    std::mutex queueMutex_;
    std::condition_variable queueCondition_;
    
    // 任务状态缓存
    std::unordered_map<std::string, GenerationTask> taskCache_;
    std::mutex cacheMutex_;
    
    // 工作线程
    std::thread workerThread_;
    bool isRunning_;
    
    // 依赖
    std::shared_ptr<GenerationTaskRepository> taskRepository_;
    
    // 工作线程函数
    void workerLoop();
    
    // 处理单个任务
    void processTask(const std::string &taskId);
    
    // 调用 AI Service
    bool callAiService(const std::string &taskId, const std::string &prompt);
};

// services/GenerationService.cc
#include "GenerationService.h"
#include "../utils/Logger.h"
#include "../dto/GenerationDto.h"
#include <chrono>

GenerationService::GenerationService()
    : isRunning_(false)
    , taskRepository_(std::make_shared<GenerationTaskRepository>()) {
    startWorker();
}

GenerationService::~GenerationService() {
    stopWorker();
}

std::string GenerationService::createTask(const GenerationRequest &request) {
    // 1. 生成任务 ID
    std::string taskId = generateUuid();
    
    // 2. 创建任务记录
    GenerationTask task;
    task.id = taskId;
    task.userId = request.userId;
    task.bookId = request.bookId;
    task.pageId = request.pageId;
    task.originalText = request.text;
    task.status = "pending";
    task.createdAt = getCurrentTimestamp();
    
    // 3. 保存到数据库
    taskRepository_->create(task);
    
    // 4. 加入任务队列
    {
        std::lock_guard<std::mutex> lock(queueMutex_);
        taskQueue_.push(taskId);
    }
    queueCondition_.notify_one();
    
    LOG_INFO << "任务已创建: " << taskId;
    
    return taskId;
}

std::optional<GenerationTask> GenerationService::getTaskStatus(const std::string &taskId) {
    // 1. 先从缓存获取
    {
        std::lock_guard<std::mutex> lock(cacheMutex_);
        auto it = taskCache_.find(taskId);
        if (it != taskCache_.end()) {
            return it->second;
        }
    }
    
    // 2. 从数据库获取
    return taskRepository_->findById(taskId);
}

void GenerationService::startWorker() {
    if (isRunning_) return;
    
    isRunning_ = true;
    workerThread_ = std::thread(&GenerationService::workerLoop, this);
    
    LOG_INFO << "工作线程已启动";
}

void GenerationService::stopWorker() {
    isRunning_ = false;
    queueCondition_.notify_all();
    
    if (workerThread_.joinable()) {
        workerThread_.join();
    }
    
    LOG_INFO << "工作线程已停止";
}

void GenerationService::workerLoop() {
    while (isRunning_) {
        std::string taskId;
        
        // 等待任务
        {
            std::unique_lock<std::mutex> lock(queueMutex_);
            queueCondition_.wait(lock, [this] {
                return !taskQueue_.empty() || !isRunning_;
            });
            
            if (!isRunning_ && taskQueue_.empty()) {
                break;
            }
            
            if (!taskQueue_.empty()) {
                taskId = taskQueue_.front();
                taskQueue_.pop();
            }
        }
        
        if (!taskId.empty()) {
            processTask(taskId);
        }
    }
}

void GenerationService::processTask(const std::string &taskId) {
    LOG_INFO << "开始处理任务: " << taskId;
    
    // 1. 获取任务
    auto task = taskRepository_->findById(taskId);
    if (!task) {
        LOG_ERROR << "任务不存在: " << taskId;
        return;
    }
    
    // 2. 更新状态为处理中
    task->status = "processing";
    task->updatedAt = getCurrentTimestamp();
    taskRepository_->update(*task);
    
    {
        std::lock_guard<std::mutex> lock(cacheMutex_);
        taskCache_[taskId] = *task;
    }
    
    // 3. 文本分析
    try {
        std::string parsedData = analyzeText(task->originalText);
        
        // 4. 优化 Prompt
        std::string prompt = optimizePrompt(parsedData);
        
        // 5. 调用 AI Service 生成图像
        bool success = callAiService(taskId, prompt);
        
        if (success) {
            // 6. 更新状态为完成
            task->status = "completed";
            task->prompt = prompt;
            task->completedAt = getCurrentTimestamp();
        } else {
            task->status = "failed";
            task->errorMessage = "AI 生成失败";
        }
    } catch (const std::exception &e) {
        task->status = "failed";
        task->errorMessage = e.what();
    }
    
    task->updatedAt = getCurrentTimestamp();
    taskRepository_->update(*task);
    
    {
        std::lock_guard<std::mutex> lock(cacheMutex_);
        taskCache_[taskId] = *task;
    }
    
    LOG_INFO << "任务处理完成: " << taskId << " 状态: " << task->status;
}

bool GenerationService::callAiService(const std::string &taskId, const std::string &prompt) {
    // 调用 Python AI Service
    // 这里使用 HTTP 客户端调用
    try {
        HttpClient client;
        
        // 构建请求
        nlohmann::json requestBody = {
            {"prompt", prompt},
            {"width", 1024},
            {"height", 1024}
        };
        
        // 发送请求
        auto response = client.post(
            "http://localhost:8000/api/v1/generation",
            requestBody.dump(),
            "application/json"
        );
        
        if (response.statusCode == 200) {
            auto result = nlohmann::json::parse(response.body);
            
            // 保存生成结果
            std::string imageUrl = result["image_url"];
            saveGenerationResult(taskId, imageUrl);
            
            return true;
        }
        
        return false;
    } catch (const std::exception &e) {
        LOG_ERROR << "调用 AI Service 失败: " << e.what();
        return false;
    }
}
```

#### 3.1.2 前端轮询实现

```typescript
// src/hooks/useGeneration.ts
import { useState, useEffect, useCallback, useRef } from 'react';
import { generationApi } from '../api/generation';
import { TaskStatus, TaskResult } from '../types/generation';

interface UseGenerationReturn {
  loading: boolean;
  taskResult: TaskResult | null;
  createTask: (request: GenerationRequest) => Promise<TaskResult>;
  reset: () => void;
}

export const useGeneration = (): UseGenerationReturn => {
  const [loading, setLoading] = useState(false);
  const [taskResult, setTaskResult] = useState<TaskResult | null>(null);
  const [taskId, setTaskId] = useState<string | null>(null);
  const pollingRef = useRef<NodeJS.Timeout | null>(null);
  
  // 轮询任务状态
  useEffect(() => {
    if (!taskId) return;
    
    const pollTask = async () => {
      try {
        const response = await generationApi.getTaskStatus(taskId);
        const result = response.data.data;
        
        setTaskResult(result);
        
        if (result.status === TaskStatus.COMPLETED || result.status === TaskStatus.FAILED) {
          setLoading(false);
          if (pollingRef.current) {
            clearInterval(pollingRef.current);
            pollingRef.current = null;
          }
        }
      } catch (error) {
        console.error('轮询任务状态失败:', error);
      }
    };
    
    // 立即执行一次
    pollTask();
    
    // 每 2 秒轮询一次
    pollingRef.current = setInterval(pollTask, 2000);
    
    return () => {
      if (pollingRef.current) {
        clearInterval(pollingRef.current);
      }
    };
  }, [taskId]);
  
  // 创建任务
  const createTask = useCallback(async (request: GenerationRequest): Promise<TaskResult> => {
    setLoading(true);
    setTaskResult(null);
    
    try {
      const response = await generationApi.createTask(request);
      const newTaskId = response.data.data.task_id;
      setTaskId(newTaskId);
      
      // 返回初始状态
      return response.data.data;
    } catch (error) {
      setLoading(false);
      throw error;
    }
  }, []);
  
  // 重置状态
  const reset = useCallback(() => {
    setLoading(false);
    setTaskResult(null);
    setTaskId(null);
    if (pollingRef.current) {
      clearInterval(pollingRef.current);
      pollingRef.current = null;
    }
  }, []);
  
  return {
    loading,
    taskResult,
    createTask,
    reset,
  };
};
```

#### 3.2.1 Redis 队列实现

```cpp
// services/RedisTaskQueue.h
#pragma once
#include <string>
#include <hiredis/hiredis.h>

class RedisTaskQueue {
public:
    RedisTaskQueue(const std::string &host, int port);
    ~RedisTaskQueue();
    
    // 入队
    bool enqueue(const std::string &taskId);
    
    // 出队
    std::string dequeue(int timeout = 0);
    
    // 获取队列长度
    size_t length();
    
    // 清空队列
    void clear();

private:
    redisContext *context_;
    std::string queueName_ = "generation_tasks";
};

// services/RedisTaskQueue.cc
#include "RedisTaskQueue.h"

RedisTaskQueue::RedisTaskQueue(const std::string &host, int port) {
    context_ = redisConnect(host.c_str(), port);
    if (context_ == NULL || context_->err) {
        throw std::runtime_error("Redis 连接失败");
    }
}

RedisTaskQueue::~RedisTaskQueue() {
    if (context_) {
        redisFree(context_);
    }
}

bool RedisTaskQueue::enqueue(const std::string &taskId) {
    redisReply *reply = (redisReply *)redisCommand(
        context_,
        "LPUSH %s %s",
        queueName_.c_str(),
        taskId.c_str()
    );
    
    bool success = (reply != NULL && reply->type == REDIS_REPLY_INTEGER);
    freeReplyObject(reply);
    
    return success;
}

std::string RedisTaskQueue::.dequeue(int timeout) {
    redisReply *reply = (redisReply *)redisCommand(
        context_,
        "BRPOP %s %d",
        queueName_.c_str(),
        timeout
    );
    
    std::string taskId;
    if (reply != NULL && reply->type == REDIS_REPLY_ARRAY && reply->elements == 2) {
        taskId = reply->element[1]->str;
    }
    
    freeReplyObject(reply);
    return taskId;
}

size_t RedisTaskQueue::length() {
    redisReply *reply = (redisReply *)redisCommand(
        context_,
        "LLEN %s",
        queueName_.c_str()
    );
    
    size_t length = 0;
    if (reply != NULL && reply->type == REDIS_REPLY_INTEGER) {
        length = reply->integer;
    }
    
    freeReplyObject(reply);
    return length;
}
```

```typescript
// src/stores/generationStore.ts
import { create } from 'zustand';
import { TaskStatus, TaskResult } from '../types/generation';

interface GenerationState {
  // 任务列表
  tasks: Map<string, TaskResult>;
  
  // 当前任务
  currentTaskId: string | null;
  
  // 方法
  addTask: (taskId: string, task: TaskResult) => void;
  updateTask: (taskId: string, task: Partial<TaskResult>) => void;
  removeTask: (taskId: string) => void;
  setCurrentTask: (taskId: string | null) => void;
  
  // 获取任务
  getTask: (taskId: string) => TaskResult | undefined;
  getTaskStatus: (taskId: string) => TaskStatus | undefined;
}

export const useGenerationStore = create<GenerationState>()((set, get) => ({
  tasks: new Map(),
  currentTaskId: null,
  
  addTask: (taskId, task) => set((state) => {
    const newTasks = new Map(state.tasks);
    newTasks.set(taskId, task);
    return { tasks: newTasks };
  }),
  
  updateTask: (taskId, updates) => set((state) => {
    const newTasks = new Map(state.tasks);
    const existingTask = newTasks.get(taskId);
    if (existingTask) {
      newTasks.set(taskId, { ...existingTask, ...updates });
    }
    return { tasks: newTasks };
  }),
  
  removeTask: (taskId) => set((state) => {
    const newTasks = new Map(state.tasks);
    newTasks.delete(taskId);
    return { tasks: newTasks };
  }),
  
  setCurrentTask: (taskId) => set({ currentTaskId: taskId }),
  
  getTask: (taskId) => get().tasks.get(taskId),
  
  getTaskStatus: (taskId) => {
    const task = get().tasks.get(taskId);
    return task?.status;
  },
}));
```

## 9. 重试机制

### 9.1 客户端重试

```typescript
// 自动重试
const retryRequest = async (fn: () => Promise<any>, maxRetries = 3) => {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (error) {
      if (i === maxRetries - 1) throw error;
      await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)));
    }
  }
};
```

### 9.2 服务端重试

```cpp
// 任务重试
bool GenerationService::retryTask(const std::string &taskId) {
    auto task = taskRepository_->findById(taskId);
    if (!task || task->status != "failed") {
        return false;
    }
    
    task->status = "pending";
    task->retryCount++;
    task->updatedAt = getCurrentTimestamp();
    taskRepository_->update(*task);
    
    // 重新加入队列
    {
        std::lock_guard<std::mutex> lock(queueMutex_);
        taskQueue_.push(taskId);
    }
    queueCondition_.notify_one();
    
    return true;
}
```

## 10. 并发控制

```cpp
// 控制并发
const int MAX_CONCURRENT_TASKS = 3;

void GenerationService::processTask(const std::string &taskId) {
    // 获取当前处理中的任务数
    size_t processingCount = taskRepository_->countByStatus("processing");
    
    if (processingCount >= MAX_CONCURRENT_TASKS) {
        // 等待一段时间后重试
        std::this_thread::sleep_for(std::chrono::seconds(5));
        
        // 重新入队
        std::lock_guard<std::mutex> lock(queueMutex_);
        taskQueue_.push(taskId);
        return;
    }
    
    // 处理任务...
}
```
