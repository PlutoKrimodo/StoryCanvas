### 4.2 自动化测试脚本

```python
# tests/api/run_tests.py
import yaml
import requests
from typing import Dict, Any

class APITestRunner:
    def __init__(self, base_url: str, token: str = None):
        self.base_url = base_url
        self.token = token
        self.session = requests.Session()
        
        if token:
            self.session.headers["Authorization"] = f"Bearer {token}"
    
    def run_test(self, test_case: Dict[str, Any]) -> bool:
        """运行单个测试用例"""
        request = test_case["request"]
        expected = test_case["expected"]
        
        # 发送请求
        url = f"{self.base_url}{request['url']}"
        response = self.session.request(
            method=request["method"],
            url=url,
            json=request.get("body"),
            headers=request.get("headers", {})
        )
        
        # 验证状态码
        if response.status_code != expected["status"]:
            print(f"❌ {test_case['name']}: 状态码不匹配")
            print(f"   预期: {expected['status']}, 实际: {response.status_code}")
            return False
        
        # 验证响应体
        response_data = response.json()
        for key, value in expected.get("body", {}).items():
            actual_value = self._get_nested_value(response_data, key)
            if actual_value != value:
                print(f"❌ {test_case['name']}: 响应体不匹配")
                print(f"   {key}: 预期 {value}, 实际 {actual_value}")
                return False
        
        print(f"✅ {test_case['name']}")
        return True
    
    def _get_nested_value(self, data: Dict, key: str):
        """获取嵌套值"""
        keys = key.split(".")
        for k in keys:
            if isinstance(data, dict):
                data = data.get(k)
            else:
                return None
        return data
    
    def run_all_tests(self, test_file: str):
        """运行所有测试用例"""
        with open(test_file, 'r', encoding='utf-8') as f:
            test_suite = yaml.safe_load(f)
        
        passed = 0
        failed = 0
        
        for module, tests in test_suite.items():
            print(f"\n📋 {module}")
            for test_case in tests:
                if self.run_test(test_case):
                    passed += 1
                else:
                    failed += 1
        
        print(f"\n📊 测试结果: {passed} 通过, {failed} 失败")
        return failed == 0

if __name__ == "__main__":
    runner = APITestRunner("http://localhost:8080")
    runner.run_all_tests("tests/api/test_cases.yaml")
```

| 修改密码后 Token 失效 | 修改密码后用旧 Token 访问 | 401 |
| 限流 | 1 分钟内发起超过 100 次匿名请求 | 429 |
| 凭据无硬编码 | 全库检索源码与配置 | 无明文密钥、口令、API Key |
